107: func (m *Module) openShift(ctx router.Context) {
108: 	var args OpenShiftArgs
109: 	if err := ctx.Decode(&args); err != nil {
110: 		ctx.WriteStatus(400); ctx.Write([]byte("bad request"))
111: 		return
112: 	}
113:
114: 	if args.TenantId == "" || args.CashierId == "" || args.Currency == "" {
115: 		ctx.WriteStatus(400); ctx.Write([]byte("bad request"))
116: 		return
117: 	}
118:
119: 	qb := m.db.Query(&Shift{})
120: 	qb.Where(Shift_.TenantId).Eq(args.TenantId)
121: 	qb.Where(Shift_.CashierId).Eq(args.CashierId)
122: 	qb.Where(Shift_.Status).Eq(ShiftOpen)
123:
124: 	existing, err := ReadAllShift(qb)
125: 	if err != nil {
126: 		ctx.WriteStatus(500); ctx.Write([]byte("internal error"))
127: 		return
128: 	}
129: 	if len(existing) > 0 {
130: 		ctx.WriteStatus(409); ctx.Write([]byte("conflict"))
131: 		return
132: 	}
133:
134: 	now := time.NowUnix()
135:
136: 	shift := &Shift{
137: 		Id:          m.deps.IDs.NewID(),
138: 		TenantId:    args.TenantId,
139: 		CashierId:   args.CashierId,
140: 		Status:      ShiftOpen,
141: 		OpeningCash: args.OpeningCash,
142: 		Currency:    args.Currency,
143: 		OpenedAt:    now,
144: 		UpdatedAt:   time.NowNano(),
145: 	}
146:
147: 	if err := shift.Validate(model.ActionCreate); err != nil {
148: 		ctx.WriteStatus(400); ctx.Write([]byte("bad request"))
149: 		return
150: 	}
151:
152: 	if err := m.db.Create(shift); err != nil {
153: 		ctx.WriteStatus(500); ctx.Write([]byte("internal error"))
154: 		return
155: 	}
156:
157: 	_ = ctx.Encode(shift)
158: }

160: func (m *Module) closeShift(ctx router.Context) {
161: 	var args CloseShiftArgs
162: 	if err := ctx.Decode(&args); err != nil {
163: 		ctx.WriteStatus(400); ctx.Write([]byte("bad request"))
164: 		return
165: 	}
166:
167: 	if args.TenantId == "" || args.Id == "" {
168: 		ctx.WriteStatus(400); ctx.Write([]byte("bad request"))
169: 		return
170: 	}
171:
172: 	shift := &Shift{}
173: 	qb := m.db.Query(shift)
174: 	qb.Where(Shift_.TenantId).Eq(args.TenantId)
175: 	qb.Where(Shift_.Id).Eq(args.Id)
176:
177: 	_, err := ReadOneShift(qb, shift)
178: 	if err != nil {
179: 		if orm.IsNotFound(err) {
180: 			ctx.WriteStatus(404); ctx.Write([]byte("not found"))
181: 			return
182: 		}
183: 		ctx.WriteStatus(500); ctx.Write([]byte("internal error"))
184: 		return
185: 	}
186:
187: 	if shift.Status == ShiftClosed {
188: 		ctx.WriteStatus(409); ctx.Write([]byte("conflict"))
189: 		return
190: 	}
191:
192: 	now := time.NowUnix()
193:
194: 	factQb := m.db.Query(&StoredFact{})
195: 	factQb.Where(StoredFact_.TenantId).Eq(args.TenantId)
196: 	factQb.Where(StoredFact_.At).Gte(int64(shift.OpenedAt))
197: 	factQb.Where(StoredFact_.At).Lte(int64(now))
198:
199: 	facts, err := ReadAllStoredFact(factQb)
200: 	if err != nil {
201: 		ctx.WriteStatus(500); ctx.Write([]byte("internal error"))
202: 		return
203: 	}
204:
205: 	expected := shift.OpeningCash
206: 	for _, f := range facts {
207: 		if f.Currency != shift.Currency || f.Means != string(money.MeansCash) || f.Actor != shift.CashierId {
208: 			continue // otra moneda, tarjeta o transferencia, u otra cajera: no es efectivo de este turno
209: 		}
210: 		switch f.Kind {
211: 		case string(money.Charged):
212: 			expected += f.Amount
213: 		case string(money.Refunded):
214: 			expected -= f.Amount
215: 		}
216: 	}
217:
218: 	shift.Status = ShiftClosed
219: 	shift.ClosingCashCounted = args.ClosingCashCounted
220: 	shift.ClosingCashExpected = expected
221: 	shift.Difference = args.ClosingCashCounted - expected
222: 	shift.ClosedAt = now
223: 	shift.UpdatedAt = time.NowNano()
224:
225: 	// We validate only the updated fields via UpdateFields, so skip Validate entirely. UpdateFields will validate the specific fields.
226:
227: 	if err := m.db.UpdateFields(shift, []string{Shift_.Status, Shift_.ClosingCashCounted, Shift_.ClosingCashExpected, Shift_.Difference, Shift_.ClosedAt, Shift_.UpdatedAt}, orm.Eq(Shift_.Id, args.Id), orm.Eq(Shift_.TenantId, args.TenantId)); err != nil {
228: 		ctx.WriteStatus(500); ctx.Write([]byte("internal error"))
229: 		return
230: 	}
231:
232: 	_ = ctx.Encode(shift)
233: }

74: // Item implementa view.Itemizer para StoredFact.

76: 	return view.Item{
77: 		ID:          f.Id,

83: // NewView construye el Presenter de consulta de hechos de caja.
84: func NewView(caller router.Caller, opts ...view.Option) view.Presenter {
85: 	allOpts := append([]view.Option{view.WithTitle(titleCashManager)}, opts...)
86: 	return view.New(&factLister{caller: caller}, &StoredFact{}, allOpts...)
87: }
88:
89: type factLister struct {
90: 	caller router.Caller
91: }
92:
93: func (l *factLister) List(done func([]model.Model, error)) {
94: 	out := &StoredFactList{}
95: 	l.caller.Call(ModelName+"."+OpListStoredFacts, nil, out, func(err error) {
96: 		if err != nil {
97: 			done(nil, err)
98: 			return
99: 		}
100: 		rows := make([]model.Model, len(*out))
101: 		for i, f := range *out {
102: 			rows[i] = f
103: 		}
104: 		done(rows, nil)
105: 	})
106: }
107: ```
